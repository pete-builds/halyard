# Double-click a tab, type a name, press Enter: the name lands on the tab
# and the shell gets none of the keys.
sleep 1500
resize 900 500
sleep 300
action new_tab
sleep 800
mousedown 40 16
mouseup 40 16
mousedown 40 16 clicks=2
mouseup 40 16 clicks=2
sleep 300
type Build
key enter
sleep 600
dump
expect names [Some("Build"), None]
reject $ Build
reject not found
quit
