# The keyboard parks when a renamed tab's shell exits mid-name. A click on
# another tab's close button (which stops the event) still counts as the user
# choosing: afterwards typing reaches the remaining shell.
sleep 1500
resize 900 500
sleep 300
action new_tab
sleep 600
action new_tab
sleep 800
type sleep 3; exit
key enter
sleep 300
action rename_tab
sleep 300
type abc
sleep 4500
dump
expect tabs 2
expect parked true
