# Rename tab 2, then click tab 1's close button: the name saves, tab 1
# closes, and typing reaches the remaining shell (the keyboard must not be
# left waiting on the window).
sleep 1500
resize 900 500
sleep 300
action new_tab
sleep 800
action rename_tab
sleep 300
type Logs
mousedown 106 16
mouseup 106 16
sleep 800
type echo after-$((2+2))
key enter
sleep 600
dump
expect tabs 1 names [Some("Logs")]
expect parked false
expect after-4
quit
