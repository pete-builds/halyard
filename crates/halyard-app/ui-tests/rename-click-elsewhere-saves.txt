# A click elsewhere while a tab name is being typed saves it: on another
# tab (which then has the keyboard), and on a pane.
sleep 1500
resize 900 500
sleep 300
action new_tab
sleep 800
mousedown 40 16
mouseup 40 16
mousedown 40 16 clicks=2
mouseup 40 16 clicks=2
sleep 300
type Logs
mousedown 140 16
mouseup 140 16
sleep 400
dump
expect names [Some("Logs"), None] active 1
type echo second-$((2+3))
key enter
sleep 600
dump
expect second-5
action rename_tab
sleep 300
type Build
mousedown 300 250
mouseup 300 250
sleep 400
dump
expect names [Some("Logs"), Some("Build")]
expect rename (tab, text, focused) None
type echo pane-$((3+4))
key enter
sleep 600
dump
expect pane-7
quit
